  <input type="button" id="btn2" name="name" value=" 全不选"/>
  <input type="button" id="btn3" name="name" value=" 反选"/>
  <script>
    $(function(){
        var input=$("input[type=radio]");
        $("#btn1").click(function(){
          $.each(input,function(index,value){
            input[index].checked=true;
          })
        })

        $("#btn2").click(function(){
             $.each(input,function(index,value){
              input[index].checked=false;
             })
        })

        $("#btn3").click(function(){
           $.each(input,function(index,value){
            input[index].checked=input[index].checked==true?false:true;
           })
        })
    });
    
  </script>
