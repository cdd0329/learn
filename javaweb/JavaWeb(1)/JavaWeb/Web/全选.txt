<!DOCTYPE html>
<html>
<head>
  <script>
    function selectAll() {
      document.querySelectorAll('input[name="option"]').forEach((checkbox) => {
        checkbox.checked = true;
      });
    }

    function deselectAll() {
      document.querySelectorAll('input[name="option"]').forEach((checkbox) => {
        checkbox.checked = false;
      });
    }

  </script>
</head>
<body>
  <input type="checkbox" name="option" value="1">选项1<br>
  <input type="checkbox" name="option" value="2">选项2<br>
  <input type="checkbox" name="option" value="3">选项3<br>
  <button onclick="selectAll()">全选</button>
  <button onclick="deselectAll()">取消全选</button>
</body>
</html>
