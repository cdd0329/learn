29、假设有一个过滤器，其全限定名为：com.kzw.web.EncodingFilter，要求它能拦截
所有的请求，且可传入字符集参数charset。请在web.xml文件中注册此过滤器。
<filter>
	<filter-name>CharSetFilter</filter-name>
	<filter-class>com.kzw.web.EncodingFilter</filter-class>
	<init-param>
		<param-name>charset</param-name>
		<param-value>UTF-8</param-value>
	</init-param>
</filter>
<filter-mapping>
	<filter-name>CharSetFilter</filter-name>
	<url-pattern>/*</url-pattern>
</filter-mapping>

<html>

		function checkval() {
			var nameElm = document.getElementById('name');
			var passElm = document.getElementById('pass');

				alert('昵称不能为空');

return false
